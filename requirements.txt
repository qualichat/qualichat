# Streamlit Community Cloud reads this at the repo root and runs
# `pip install -r requirements.txt`. It bundles every dependency
# qualichat needs at runtime, including the web UI (streamlit) and
# the two spaCy language models (downloaded from the spacy-models
# GitHub releases, since Streamlit Cloud cannot run setup scripts).
#
# For local development prefer `pip install -e ".[dev]"` driven by
# pyproject.toml; this file exists primarily so Streamlit Cloud can
# build the web UI deploy.

# Pin numpy < 2 because spaCy 3.7.x ships C-extensions (thinc, blis)
# compiled against the numpy 1.x ABI. With numpy 2.x at runtime, every
# parse raises:
#   "numpy.dtype size changed, may indicate binary incompatibility.
#    Expected 96 from C header, got 88 from PyObject."
# Upgrading to spaCy 3.8+ would lift this constraint, but that is a
# wider change. For now pin numpy.
numpy>=1.26,<2

# CLI fallbacks (used by qualichat.frames internals even when driven
# from the web UI through qualichat.ui.runtime).

# without running a separate `spacy download` step at first launch)

en_core_web_sm @ https://github.com/explosion/spacy-models/releases/download/en_core_web_sm-3.7.1/en_core_web_sm-3.7.1-py3-none-any.whl
